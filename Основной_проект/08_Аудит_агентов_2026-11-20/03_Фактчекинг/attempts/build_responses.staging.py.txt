def at(j):
 return datetime.strptime(J[j]['at']['current_time'],'%Y-%m-%d %H:%M:%S UTC').replace(tzinfo=timezone.utc).astimezone(ZoneInfo('Asia/Yekaterinburg')).isoformat()
def lines(j,lo,hi,ref=None):
 text=J[j]['result']; found=[]
 if ref:
  marker='cite'+ref+''
  text=text.split(marker,1)[1].split('--------------------------------------------------------------------------------',1)[0]
 for l in text.splitlines():
  m=re.match(r'L(\d+)(?:@[^:]+)?: (.*)',l)
  if m and lo<=int(m[1])<=hi: found.append(m[2])
 return '\n'.join(found)

def align(entity,period,boundary,metric,unit):
